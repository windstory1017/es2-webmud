$ cat es2.exp

#!/usr/bin/expect

set timeout 30

set user [lindex $argv 0]

set password [lindex $argv 1]

spawn telnet es2.muds.net 4000

expect "您的使用者代號："

sleep 1;

send "$user\r";

expect "請輸入密碼﹕"

sleep 1;

send "$password\r";

expect "出口"

sleep 1;

send "quit\r";

$ ./es2.exp  <user1> <password1>

$ ./es2.exp  <user2> <password2>

$ ./es2.exp  <user3> <password3>

$ ./es2.exp  <user4> <password4>
